<%@ page language="java" contentType="text/html" %>
<%@ page import="java.text.DecimalFormat" %>
<%
    // Read form parameters
    String name = request.getParameter("consumerName");
    String consumerNo = request.getParameter("consumerNo");
    String meterNo = request.getParameter("meterNo");
    int prev = Integer.parseInt(
        request.getParameter("prevReading"));
    int curr = Integer.parseInt(
        request.getParameter("currReading"));
    String category = request.getParameter("category");

    int units = curr - prev;

    // Calculate energy charge (slab-based)
    double energyCharge = 0;
    if (category.equals("residential")) {
        if (units <= 100)
            energyCharge = units * 3.50;
        else if (units <= 300)
            energyCharge = 100*3.50 + (units-100)*5.50;
        else if (units <= 500)
            energyCharge = 100*3.50 + 200*5.50
                          + (units-300)*7.00;
        else
            energyCharge = 100*3.50 + 200*5.50
                          + 200*7.00 + (units-500)*8.50;
    } else if (category.equals("commercial")) {
        energyCharge = units * 8.00;
    } else {
        energyCharge = units * 7.50;
    }

    double fixed = category.equals("residential")
        ? 50 : category.equals("commercial") ? 200 : 350;
    double duty = energyCharge * 0.10;
    double fuel = energyCharge * 0.02;
    double total = energyCharge + fixed + duty + fuel;

    DecimalFormat df = new DecimalFormat("#,##0.00");
%>
